/**
 * Marnie & Zach — website RSVPs → "Wedding Planning" Google Sheet
 *
 * What it does
 *   • Pulls every reply from the wedding website into a tab called "RSVPs".
 *   • Adds four columns to the right of the "Guest List" tab and keeps them up to date:
 *       RSVP · Adults coming · RSVP name(s) · Replied
 *   • Flags replies from people it can't find on the guest list (extras, or names to link by hand).
 *
 * It only ever writes to those four columns and to the "RSVPs" tab. Nothing else in the sheet is touched.
 *
 * Set-up: paste this whole file into Extensions → Apps Script, save, reload the sheet,
 * then choose  Wedding RSVPs → Connect to the website…
 */

const SITE_URL_DEFAULT = "https://zach-marnie-wedding.vercel.app";
const GUEST_SHEET = "Guest List";
const RSVP_SHEET = "RSVPs";
const NAME_HEADER = "Guest / Party Name";
const OUR_HEADERS = ["RSVP", "Adults coming", "RSVP name(s)", "Replied"];
const TZ = "Australia/Brisbane";
const REFRESH_MINUTES = 15;

// Layout of the RSVPs tab (this script owns the whole tab).
const R_HEAD_ROW = 7;
const R_FIRST_ROW = 8;
const R_HEADERS = ["Replied", "Name", "Attending", "Guest list match", "Note", "Also in this reply",
  "Adults in reply", "Children in reply", "Reply ID", "Timestamp"];
const R_COLS = R_HEADERS.length;

const COLOURS = { Attending: "#d9ead3", Declined: "#f4cccc", "Part attending": "#fff2cc", check: "#fce5cd", unmatched: "#fce5cd" };

/* ───────────────────────── Menu & set-up ───────────────────────── */

function onOpen() {
  SpreadsheetApp.getUi().createMenu("Wedding RSVPs")
    .addItem("Refresh now", "refreshRsvps")
    .addSeparator()
    .addItem("Connect to the website…", "connect")
    .addItem("Change website address…", "changeSiteUrl")
    .addToUi();
}

function connect() {
  const ui = SpreadsheetApp.getUi();
  const ask = ui.prompt("Connect to the wedding website",
    "Paste the sheet sync key from the website's admin page (" + siteUrl_() + "/admin):", ui.ButtonSet.OK_CANCEL);
  if (ask.getSelectedButton() !== ui.Button.OK) return;
  const key = ask.getResponseText().trim();
  if (!key) return;
  PropertiesService.getDocumentProperties().setProperty("SYNC_KEY", key);

  try {
    refresh_();
  } catch (err) {
    ui.alert("Couldn't connect", String(err.message || err), ui.ButtonSet.OK);
    return;
  }

  // One timer only: remove any earlier ones before adding it.
  ScriptApp.getProjectTriggers().forEach(function (t) {
    if (t.getHandlerFunction() === "refreshRsvps") ScriptApp.deleteTrigger(t);
  });
  ScriptApp.newTrigger("refreshRsvps").timeBased().everyMinutes(REFRESH_MINUTES).create();

  ui.alert("Connected", "RSVPs will now refresh every " + REFRESH_MINUTES +
    " minutes. Use Wedding RSVPs → Refresh now to update straight away.", ui.ButtonSet.OK);
}

function changeSiteUrl() {
  const ui = SpreadsheetApp.getUi();
  const ask = ui.prompt("Website address", "Currently " + siteUrl_() + "\nEnter the new address:", ui.ButtonSet.OK_CANCEL);
  if (ask.getSelectedButton() !== ui.Button.OK) return;
  const url = ask.getResponseText().trim().replace(/\/+$/, "");
  if (!/^https:\/\//.test(url)) { ui.alert("The address needs to start with https://"); return; }
  PropertiesService.getDocumentProperties().setProperty("SITE_URL", url);
}

function siteUrl_() {
  return PropertiesService.getDocumentProperties().getProperty("SITE_URL") || SITE_URL_DEFAULT;
}

function fetchFeed_() {
  const key = PropertiesService.getDocumentProperties().getProperty("SYNC_KEY");
  if (!key) throw new Error("Not connected yet. Choose Wedding RSVPs → Connect to the website…");
  const res = UrlFetchApp.fetch(siteUrl_() + "/api/sheet-feed", {
    headers: { Authorization: "Bearer " + key }, muteHttpExceptions: true,
  });
  const code = res.getResponseCode();
  if (code === 401) throw new Error("The website didn't accept the sync key. Copy it again from the admin page and reconnect.");
  if (code !== 200) throw new Error("The website returned an error (" + code + "). Try again in a minute.");
  const body = JSON.parse(res.getContentText());
  if (!body || !Array.isArray(body.rsvps)) throw new Error("The website sent something unexpected.");
  return body.rsvps;
}

/* ───────────────────────── Main jobs ───────────────────────── */

/** Menu item and 15-minute timer. From the menu a problem is shown on screen; on the timer it is noted on the RSVPs tab. */
function refreshRsvps(e) {
  const onTimer = !!(e && e.triggerUid);
  try {
    refresh_();
  } catch (err) {
    const message = String(err.message || err);
    if (!onTimer) { SpreadsheetApp.getUi().alert("Couldn't refresh", message, SpreadsheetApp.getUi().ButtonSet.OK); return; }
    const sheet = SpreadsheetApp.getActive().getSheetByName(RSVP_SHEET);
    if (sheet) sheet.getRange(3, 1).setValue("Last automatic refresh failed: " + message);
  }
}

/** Pulls replies from the website, links new ones to the guest list, and rewrites both tabs. */
function refresh_() {
  const lock = LockService.getDocumentLock();
  if (!lock.tryLock(20000)) return;
  try {
    const ss = SpreadsheetApp.getActive();
    const people = flattenPeople(fetchFeed_());
    const guestSheet = guestSheet_(ss);
    const layout = ensureGuestColumns_(guestSheet);
    const rsvpSheet = rsvpSheet_(ss);

    // Anyone already listed in the RSVPs tab has had their one automatic match attempt.
    const seen = new Set(readPeople_(rsvpSheet).map(function (p) { return p.key; }));
    const changed = autoLink(layout.guests, people, seen);
    changed.forEach(function (g) {
      guestSheet.getRange(g.r + 1, layout.cols.link + 1).setValue(g.links.join(", "));
    });

    writeResults_(guestSheet, layout, rsvpSheet, people, true);
  } finally {
    lock.releaseLock();
  }
}

/** Re-checks the guest list straight away when someone edits the "RSVP name(s)" column by hand. */
function onEdit(e) {
  if (!e || !e.range) return;
  const sheet = e.range.getSheet();
  if (sheet.getName().trim().toLowerCase() !== GUEST_SHEET.toLowerCase()) return;
  const grid = sheet.getDataRange().getValues();
  const parsed = parseGuestGrid(grid);
  if (parsed.cols.link < 0) return;
  const linkCol = parsed.cols.link + 1;
  if (e.range.getColumn() > linkCol || e.range.getLastColumn() < linkCol) return;
  const rsvpSheet = SpreadsheetApp.getActive().getSheetByName(RSVP_SHEET);
  if (!rsvpSheet) return;
  writeResults_(sheet, parsed, rsvpSheet, readPeople_(rsvpSheet), false);
}

/* ───────────────────────── Sheet reading & writing ───────────────────────── */

function guestSheet_(ss) {
  const found = ss.getSheets().filter(function (s) { return s.getName().trim().toLowerCase() === GUEST_SHEET.toLowerCase(); })[0];
  if (!found) throw new Error('Couldn\'t find a tab called "' + GUEST_SHEET + '".');
  return found;
}

/** Finds the four RSVP columns on the Guest List tab, adding them to the right of the table the first time. */
function ensureGuestColumns_(sheet) {
  let parsed = parseGuestGrid(sheet.getDataRange().getValues());
  if (!parsed.headerRows.length) throw new Error('Couldn\'t find a "' + NAME_HEADER + '" heading on the ' + GUEST_SHEET + " tab.");
  if (parsed.cols.rsvp >= 0 && parsed.cols.coming >= 0 && parsed.cols.link >= 0 && parsed.cols.replied >= 0) return parsed;

  const first = sheet.getLastColumn() + 1;
  const needed = first + OUR_HEADERS.length - 1;
  if (sheet.getMaxColumns() < needed) sheet.insertColumnsAfter(sheet.getMaxColumns(), needed - sheet.getMaxColumns());
  parsed.headerRows.forEach(function (h) {
    sheet.getRange(h + 1, first, 1, OUR_HEADERS.length).setValues([OUR_HEADERS]).setFontWeight("bold");
  });
  sheet.setColumnWidth(first, 120);
  sheet.setColumnWidth(first + 1, 110);
  sheet.setColumnWidth(first + 2, 240);
  sheet.setColumnWidth(first + 3, 90);
  parsed = parseGuestGrid(sheet.getDataRange().getValues());
  return parsed;
}

function rsvpSheet_(ss) {
  let sheet = ss.getSheetByName(RSVP_SHEET);
  if (sheet) return sheet;
  sheet = ss.insertSheet(RSVP_SHEET, ss.getSheets().length);
  sheet.getRange(1, 1).setValue("RSVPs from the website").setFontWeight("bold").setFontSize(14);
  sheet.getRange(R_HEAD_ROW, 1, 1, R_COLS).setValues([R_HEADERS]).setFontWeight("bold");
  sheet.setFrozenRows(R_HEAD_ROW);
  [150, 200, 90, 200, 260, 240, 110, 120, 110, 110].forEach(function (w, i) { sheet.setColumnWidth(i + 1, w); });
  return sheet;
}

/** Reads the people currently listed on the RSVPs tab (newest first, as written by this script). */
function readPeople_(sheet) {
  const last = sheet.getLastRow();
  if (last < R_FIRST_ROW) return [];
  const rows = sheet.getRange(R_FIRST_ROW, 1, last - R_FIRST_ROW + 1, R_COLS).getValues();
  return peopleFromRows(rows);
}

function writeResults_(guestSheet, layout, rsvpSheet, people, fetched) {
  const result = compute(layout.guests, people);
  const fmt = function (iso, pattern) { return iso ? Utilities.formatDate(new Date(iso), TZ, pattern) : ""; };

  // ── Guest List: status, adults coming and reply date for every guest row.
  if (layout.guests.length) {
    const top = layout.guests[0].r, bottom = layout.guests[layout.guests.length - 1].r, n = bottom - top + 1;
    const ours = Object.keys(COLOURS).map(function (k) { return COLOURS[k]; });
    const write = function (col, pick, coloured) {
      const range = guestSheet.getRange(top + 1, col + 1, n, 1);
      const values = range.getValues();
      const fills = coloured ? range.getBackgrounds() : null;
      result.guests.forEach(function (g) {
        const i = g.r - top;
        values[i][0] = pick(g);
        if (!fills) return;
        // Colour the status; only ever clear a colour this script put there.
        if (g.status) fills[i][0] = COLOURS[g.status] || COLOURS.check;
        else if (ours.indexOf(String(fills[i][0]).toLowerCase()) >= 0) fills[i][0] = null;
      });
      range.setValues(values);
      if (fills) range.setBackgrounds(fills);
    };
    // Plain text, so a date like "9 Oct" or a name isn't reinterpreted by Sheets.
    guestSheet.getRange(top + 1, layout.cols.replied + 1, n, 1).setNumberFormat("@");
    write(layout.cols.rsvp, function (g) { return g.status; }, true);
    write(layout.cols.coming, function (g) { return g.coming; }, false);
    write(layout.cols.replied, function (g) { return fmt(g.replied, "d MMM"); }, false);
  }

  // ── RSVPs tab: summary, then one row per person.
  const s = result.summary;
  if (fetched) {
    rsvpSheet.getRange(2, 1).setValue("Last refreshed " + Utilities.formatDate(new Date(), TZ, "d MMM yyyy, h:mm a") +
      " · updates every " + REFRESH_MINUTES + " minutes");
    rsvpSheet.getRange(3, 1).clearContent();
  }
  rsvpSheet.getRange(4, 1, 2, 6).setValues([
    ["People replied", "Adults attending", "Children attending", "Declined", "Not matched to guest list", "Guests still to reply"],
    [s.replied, s.attending, s.kids, s.declined, s.unmatched, s.awaiting],
  ]);
  rsvpSheet.getRange(4, 1, 1, 6).setFontWeight("bold");

  const old = rsvpSheet.getLastRow();
  if (old >= R_FIRST_ROW) {
    rsvpSheet.getRange(R_FIRST_ROW, 1, old - R_FIRST_ROW + 1, R_COLS).clearContent().setBackground(null);
  }
  if (result.people.length) {
    const rows = result.people.map(function (p) {
      return [fmt(p.date, "d MMM yyyy, h:mm a"), p.name, p.attending ? "Yes" : "No", p.match, p.note,
        p.party.join(", "), p.adults, p.kids, p.replyId, p.date];
    });
    const fills = result.people.map(function (p) {
      const row = new Array(R_COLS).fill(null);
      if (!p.matched && !p.superseded) row[3] = COLOURS.unmatched;
      return row;
    });
    rsvpSheet.getRange(R_FIRST_ROW, 1, rows.length, 6).setNumberFormat("@");
    rsvpSheet.getRange(R_FIRST_ROW, 9, rows.length, 2).setNumberFormat("@");
    rsvpSheet.getRange(R_FIRST_ROW, 1, rows.length, R_COLS).setValues(rows).setBackgrounds(fills);
  }
}

/* ───────────────────────── Matching logic (no sheet access below here) ───────────────────────── */

const TITLES = ["uncle", "aunty", "auntie", "aunt", "grandma", "grandpa", "nana", "nan", "pop", "dad", "mum", "mom",
  "mr", "mrs", "ms", "miss", "dr", "fr", "father"];

function norm(s) {
  return String(s == null ? "" : s).normalize("NFD").replace(/[̀-ͯ]/g, "").toLowerCase()
    .replace(/[^\p{L}\p{N}]+/gu, " ").trim();
}
function splitNames(s) {
  return String(s == null ? "" : s).split(/[,;\n]+/).map(function (x) { return x.trim(); }).filter(Boolean);
}
function num(v) { const n = Number(v); return isFinite(n) ? n : 0; }

/** Turns the Guest List grid into a list of guest rows plus the positions of the columns we use. */
function parseGuestGrid(grid) {
  const is = function (v, text) { return String(v == null ? "" : v).trim().toLowerCase() === text.toLowerCase(); };
  const headerRows = [];
  let nameCol = -1;
  grid.forEach(function (row, r) {
    const c = row.findIndex(function (v) { return is(v, NAME_HEADER); });
    if (c >= 0) { headerRows.push(r); if (nameCol < 0) nameCol = c; }
  });
  const cols = { name: nameCol, adults: -1, kids: -1, under6: -1, rsvp: -1, coming: -1, link: -1, replied: -1 };
  const guests = [];
  if (!headerRows.length) return { headerRows: headerRows, cols: cols, guests: guests };

  const head = grid[headerRows[0]];
  const find = function (text) { return head.findIndex(function (v) { return is(v, text); }); };
  cols.adults = find("Full Paying Adults");
  cols.kids = find("Kids Meals");
  cols.under6 = find("Non-Paying (<6)");
  cols.rsvp = find(OUR_HEADERS[0]);
  cols.coming = find(OUR_HEADERS[1]);
  cols.link = find(OUR_HEADERS[2]);
  cols.replied = find(OUR_HEADERS[3]);

  headerRows.forEach(function (h) {
    const section = h > 0 ? String(grid[h - 1][nameCol] || "").trim() : "";
    for (let r = h + 1; r < grid.length; r++) {
      const raw = String(grid[r][nameCol] == null ? "" : grid[r][nameCol]).trim();
      if (!raw || /subtotal|grand total|^total/i.test(raw) || is(raw, NAME_HEADER)) break;
      const plus = /\+\s*(\d+)/.exec(raw);
      const adults = cols.adults >= 0 ? num(grid[r][cols.adults]) : 1;
      const kids = (cols.kids >= 0 ? num(grid[r][cols.kids]) : 0) + (cols.under6 >= 0 ? num(grid[r][cols.under6]) : 0);
      guests.push({
        r: r, name: raw, section: section, adults: adults,
        kidRow: adults === 0 && kids > 0,
        plus: plus ? Number(plus[1]) : 0,
        capacity: Math.max(1, adults, plus ? 1 + Number(plus[1]) : 0),
        tokens: norm(raw.replace(/\+\s*\d+.*$/, "")).split(" ").filter(function (t) { return t && TITLES.indexOf(t) < 0; }),
        links: cols.link >= 0 ? splitNames(grid[r][cols.link]) : [],
      });
    }
  });
  return { headerRows: headerRows, cols: cols, guests: guests };
}

/** One entry per named person, newest reply first. Older replies under the same name are marked superseded. */
function flattenPeople(rsvps) {
  const people = [];
  rsvps.forEach(function (r) {
    const names = (r.names || []).map(function (n) { return String(n).trim(); }).filter(Boolean);
    names.forEach(function (name) {
      people.push({
        key: r.id + "|" + norm(name), replyId: r.id, name: name, attending: r.attending === true,
        date: r.createdAt, adults: num(r.adults), kids: r.attending === true ? num(r.kids) : 0,
        party: names.filter(function (n) { return n !== name; }),
      });
    });
  });
  return markSuperseded(people);
}

function peopleFromRows(rows) {
  const people = [];
  rows.forEach(function (row) {
    const name = String(row[1] == null ? "" : row[1]).trim();
    if (!name) return;
    const date = row[9] instanceof Date ? row[9].toISOString() : String(row[9] || "");
    people.push({
      key: String(row[8]) + "|" + norm(name), replyId: String(row[8]), name: name,
      attending: String(row[2]).trim().toLowerCase() === "yes", date: date,
      adults: num(row[6]), kids: num(row[7]), party: splitNames(row[5]),
    });
  });
  return markSuperseded(people);
}

function markSuperseded(people) {
  people.sort(function (a, b) { return String(b.date).localeCompare(String(a.date)); });
  const latest = new Set();
  people.forEach(function (p) {
    const n = norm(p.name);
    p.superseded = latest.has(n);
    latest.add(n);
  });
  return people;
}

/**
 * Guest rows a person could be, best matches only.
 * A guest written as a full name needs first and last name to agree; a first-name-only guest matches on first name.
 */
function candidates(guests, name, anyCapacity) {
  const p = norm(name).split(" ").filter(Boolean);
  if (!p.length) return [];
  let best = 0, out = [];
  guests.forEach(function (g) {
    if (g.kidRow || !g.tokens.length) return;
    if (!anyCapacity && g.links.length >= g.capacity) return;
    const t = g.tokens;
    let score = 0;
    if (t.length >= 2) {
      if (p.length >= 2 && t[0] === p[0] && t[t.length - 1] === p[p.length - 1]) score = 3;
    } else if (t[0] === p[0]) {
      score = p.length === 1 ? 2 : 1;
    }
    if (!score) return;
    if (score > best) { best = score; out = [g]; } else if (score === best) out.push(g);
  });
  return out;
}

/**
 * Links people who have just replied to a guest row when there is exactly one sensible match.
 * Never touches people who were already on the RSVPs tab, so a link removed by hand stays removed.
 * Returns the guest rows whose links changed.
 */
function autoLink(guests, people, seen) {
  const linked = new Set();
  guests.forEach(function (g) { g.links.forEach(function (n) { linked.add(norm(n)); }); });
  const changed = new Set();
  const link = function (g, p) { g.links.push(p.name); linked.add(norm(p.name)); changed.add(g); };

  const fresh = people.filter(function (p) { return !seen.has(p.key) && !p.superseded; })
    .sort(function (a, b) { return String(a.date).localeCompare(String(b.date)); });

  const unplaced = [];
  fresh.forEach(function (p) {
    if (linked.has(norm(p.name))) return;
    const c = candidates(guests, p.name, false);
    if (c.length === 1) link(c[0], p); else unplaced.push(p);
  });

  // A "+ 1" guest's partner: no match of their own, but they replied together with someone on a "+ 1" row.
  unplaced.forEach(function (p) {
    if (linked.has(norm(p.name)) || candidates(guests, p.name, true).length) return;
    const mates = p.party.map(norm);
    const host = guests.filter(function (g) {
      return g.plus > 0 && g.links.length < g.capacity &&
        g.links.some(function (n) { return mates.indexOf(norm(n)) >= 0; });
    })[0];
    if (host) link(host, p);
  });
  return Array.from(changed);
}

/** Works out what to show for every guest row and every person who replied. */
function compute(guests, people) {
  const latest = new Map();
  people.forEach(function (p) { if (!p.superseded) latest.set(norm(p.name), p); });
  const owner = new Map();
  guests.forEach(function (g) { g.links.forEach(function (n) { if (!owner.has(norm(n))) owner.set(norm(n), g); }); });

  const guestOut = guests.map(function (g) {
    if (!g.links.length) return { r: g.r, status: "", coming: "", replied: "" };
    const found = g.links.map(function (n) { return latest.get(norm(n)); }).filter(Boolean);
    if (!found.length) return { r: g.r, status: "Check name", coming: "", replied: "" };
    const yes = found.filter(function (p) { return p.attending; }).length;
    let status = yes === found.length ? "Attending" : yes === 0 ? "Declined" : "Part attending";
    if (found.length < g.links.length) status += " (check name)";
    const replied = found.map(function (p) { return String(p.date); }).sort().pop();
    return { r: g.r, status: status, coming: yes, replied: replied };
  });

  const peopleOut = people.map(function (p) {
    const g = owner.get(norm(p.name));
    let match = "", note = "";
    if (p.superseded) {
      note = "Earlier reply from the same name. The newer one counts.";
      match = g ? g.name : "";
    } else if (g) {
      match = g.name;
      if (g.links.length > Math.max(1, g.adults)) note = "More names than this guest's invite";
    } else {
      match = "NOT MATCHED";
      const c = candidates(guests, p.name, true);
      const hosts = p.party.map(function (n) { return owner.get(norm(n)); }).filter(Boolean);
      if (c.length) note = "Could be: " + c.map(function (x) { return x.name + " (row " + (x.r + 1) + ")"; }).join(", ");
      else if (hosts.length) note = "Not on the guest list. Replied with " + hosts[0].name + " (row " + (hosts[0].r + 1) + ")";
      else note = "Not on the guest list";
    }
    return {
      date: p.date, name: p.name, attending: p.attending, match: match, note: note, party: p.party,
      adults: p.adults, kids: p.kids, replyId: p.replyId, matched: !!g, superseded: p.superseded,
    };
  });

  const current = people.filter(function (p) { return !p.superseded; });
  const kidsByReply = new Map();
  current.forEach(function (p) { if (p.attending) kidsByReply.set(p.replyId, p.kids); });
  let kids = 0;
  kidsByReply.forEach(function (k) { kids += k; });

  return {
    guests: guestOut, people: peopleOut,
    summary: {
      replied: current.length,
      attending: current.filter(function (p) { return p.attending; }).length,
      declined: current.filter(function (p) { return !p.attending; }).length,
      kids: kids,
      unmatched: current.filter(function (p) { return !owner.get(norm(p.name)); }).length,
      awaiting: guests.filter(function (g) { return g.adults > 0 && !g.kidRow && !g.links.length; }).length,
    },
  };
}
